from head import TEMPORAL_LAGS, TemporalHead, anomaly_logits, gated_delta, temporal_evidence
from v7_windows import overlap_forward


def mine_pairs(array, support, upper, device):
    positives = np.flatnonzero(support >= 0.60)
    order = np.argsort(-support[positives], kind="stable")
    positives = positives[order[:min(16, max(1, len(support) // 16))]]
    if not len(positives):
        return positives, np.empty(0, dtype=np.int64), np.empty(0, dtype=np.float32)

    # Require an entire neighborhood of low scores in both V7 branches.
    low = np.ones(len(upper), dtype=bool)
    padded = np.pad(upper, (4, 4), constant_values=1.)
    for offset in range(9):
        low &= padded[offset:offset + len(upper)] <= 0.35
    background = np.flatnonzero(low)
    matches = np.full(len(positives), -1, dtype=np.int64)
    confidence = np.ones(len(positives), dtype=np.float32)
    if len(background):
        features = F.normalize(torch.as_tensor(array, device=device).float(), dim=-1)
        pos_index = torch.as_tensor(positives, device=device)
        bg_index = torch.as_tensor(background, device=device)
        sim = (features[pos_index] @ features[bg_index].T).cpu().numpy()
        distance = np.abs(positives[:, None] - background)
        sim[(distance < 5) | (distance > 64)] = -np.inf
        best = sim.argmax(axis=1)
        for row in np.flatnonzero(sim[np.arange(len(best)), best] >= 0.45):
            neg = background[best[row]]
            # Limit replacement slots to half of the positives, rounded up.
            if np.count_nonzero(matches >= 0) >= (len(positives) + 1) // 2:
                break
            matches[row] = neg
            confidence[row] = 0.5 + 0.5 * (support[positives[row]] - upper[neg])
    return positives, matches, confidence


def cache_v7(model, context, indices, train, make_mask):
    pack, ids, _, abnormal, args, device, prompts, primary = context
    cache, mined = {}, []
    width = args.visual_length
    train = set(train)
    with torch.inference_mode():
        for checked, i in enumerate(indices, 1):
            key = ids[i]
            array = pack.array(pack.features[key])
            steps = len(array)
            blocks = (steps + width - 1) // width
            x = torch.zeros((blocks * width, 512), device=device)
            x[:steps] = torch.as_tensor(array, dtype=torch.float32, device=device)
            lengths = (steps - torch.arange(blocks) * width).clamp(max=width)
            _, c, a = overlap_forward(model, x.reshape(blocks, width, 512),
                                      make_mask(lengths, width).to(device), prompts, lengths, make_mask)
            c, a = (v.flatten()[:steps] for v in anomaly_logits(c, a))
            base = c if primary == "C" else a
            _, support, upper = temporal_evidence(c, a)
            if not torch.isfinite(torch.stack((c, a))).all():
                raise ValueError(f"Invalid V7 logits: {key}")
            clip = array.astype(np.float16)
            support_np = support.cpu().numpy()
            cache[int(i)] = (clip, base.cpu().numpy(), support_np)
            if i in train and abnormal[i]:
                pairs = mine_pairs(clip, support_np, upper.cpu().numpy(), device)
                mined.append({"video": key, "positive": pairs[0].tolist(),
                              "background": pairs[1].tolist(), "weight": pairs[2].tolist()})
                cache[int(i)] += (pairs,)
            if checked % 100 == 0 or checked == len(indices):
                print(f"{PREFIX}: cached V7 + CLIP {checked}/{len(indices)}", flush=True)
    (OUTPUT / "mined_pairs.json").write_text(json.dumps(mined), encoding="utf-8")
    slots = sum(sum(j >= 0 for j in row["background"]) for row in mined)
    return cache, {"videos": len(cache), "temporal_lags": TEMPORAL_LAGS,
                   "videos_with_pairs": sum(any(j >= 0 for j in row["background"]) for row in mined),
                   "unique_replacement_slots": slots}


def top_score(logits):
    return logits.topk(1 + len(logits) // 16).values.mean()


def fit_one(context, schedule, dev, cache, variant):
    _, _, _, abnormal, _, device, _, _ = context
    seed_all(SEED)
    head = TemporalHead().to(device)
    opt = torch.optim.AdamW(head.parameters(), lr=0.001)
    strength = TEMPORAL_CONFIG["max_logit_delta"] / 2
    total_pairs = replaced = 0
    history = []
    for epoch, batches in enumerate(schedule):
        rng = np.random.default_rng(SEED + epoch)
        losses = []
        for batch in batches:
            scores, positives, normals, backgrounds, weights = [], [], [], [], []
            for i in batch:
                clip, base, support, *extra = cache[int(i)]
                x = torch.as_tensor(clip, device=device)
                raw = head(x, True)
                z = torch.as_tensor(base, device=device) + gated_delta(
                    raw, torch.as_tensor(support, device=device), strength,
                )
                scores.append(top_score(z))
                if abnormal[i]:
                    pos, neg, weight = extra[0]
                    if len(pos):
                        ix = torch.as_tensor(neg.clip(min=0), device=device)
                        positives.append(raw[torch.as_tensor(pos, device=device)])
                        backgrounds.append((raw[ix], torch.as_tensor(neg >= 0, device=device)))
                        weights.append(torch.as_tensor(weight, device=device))
                else:
                    ix = rng.choice(len(raw), min(16, len(raw)), replace=False)
                    normals.append(raw[torch.as_tensor(ix, device=device)])
            labels = torch.as_tensor(abnormal[batch], device=device, dtype=torch.float32)
            loss = F.binary_cross_entropy_with_logits(torch.stack(scores), labels)
            if positives and normals:
                pos = torch.cat(positives)
                normal = torch.cat(normals)
                sampled = normal[torch.as_tensor(rng.integers(len(normal), size=len(pos)), device=device)]
                bg, valid = (torch.cat(parts) for parts in zip(*backgrounds, strict=True))
                weight = torch.cat(weights)
                if variant == "v19":
                    sampled = torch.where(valid, bg, sampled)
                    replaced += int(valid.sum())
                pair_loss = F.softplus(0.5 - pos + sampled)
                loss = loss + 0.5 * (weight * pair_loss).sum() / weight.sum()
                total_pairs += len(pos)
            if not torch.isfinite(loss):
                raise ValueError(f"Invalid V19 training loss: {variant}")
            opt.zero_grad(set_to_none=True)
            loss.backward()
            opt.step()
            losses.append(float(loss.detach()))
        history.append(float(np.mean(losses)))
        print(f"{PREFIX}/{variant}: epoch {epoch+1}, loss={history[-1]:.5f}, "
              f"pairs={total_pairs}, replaced={replaced}", flush=True)
    head.eval()
    scores, frozen_scores = [], []
    with torch.no_grad():
        for i in dev:
            clip, base, support = cache[int(i)]
            b = torch.as_tensor(base, device=device)
            raw = head(torch.as_tensor(clip, device=device), True)
            delta = gated_delta(raw, torch.as_tensor(support, device=device), strength)
            scores.append(float(top_score(b + delta)))
            frozen_scores.append(float(top_score(b)))
    path = OUTPUT / f"{variant}.pth"
    save_checkpoint(head, path)
    return {"strength": strength, "max_logit_delta": TEMPORAL_CONFIG["max_logit_delta"],
            "dev_video_ap": float(average_precision_score(abnormal[dev], scores)),
            "dev_v7_video_ap": float(average_precision_score(abnormal[dev], frozen_scores)),
            "paired_examples": total_pairs, "replaced_pairs": replaced,
            "epoch_loss": history, "checkpoint_file": path.name, "checkpoint_sha256": file_sha(path)}


def fit_variants(model, context, train, dev, make_mask):
    _, _, _, abnormal, _, _, _, _ = context
    schedule = [balanced_batches(train, abnormal, 2 if SMOKE_ONLY else HALF_BATCH, SEED + epoch)
                for epoch in range(1 if SMOKE_ONLY else TEMPORAL_CONFIG["epochs"])]
    if SMOKE_ONLY:
        schedule = [schedule[0][:2]]
    used = np.unique(np.concatenate([batch.ravel() for batch in schedule]))
    indices = np.unique(np.r_[used if SMOKE_ONLY else train, dev])
    cache, receipt = cache_v7(model, context, indices, used, make_mask)
    result = {variant: fit_one(context, schedule, dev, cache, variant) for variant in VARIANTS}
    if result["normal_only"]["paired_examples"] != result["v19"]["paired_examples"]:
        raise ValueError("V19 pair budgets differ.")
    if not result["v19"]["replaced_pairs"]:
        message = f"{PREFIX}: no same-video background pairs passed mining"
        if not SMOKE_ONLY:
            raise ValueError(message)
        print(message + " in the smoke sample.", flush=True)
    return result, receipt
